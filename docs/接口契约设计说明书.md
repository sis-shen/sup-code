 ## 2. 设计原则
 
 1. **接口优先**：层与层之间、模块与模块之间一律通过 Go interface 通信，不依赖具体实现
 2. **零侵入 Mock**：每个 interface 设计时即为 mock 友好，方便 `testify/mock` 或手写 mock
 3. **契约即文档**：接口签名本身构成契约，配合 godoc 注释即可投产，不另维护同步文档
 4. **不可变参数**：跨层传递的 DTO 使用值类型或不可变结构，避免并发修改
 5. **context.Context 贯穿**：所有涉及 I/O 或可能阻塞的方法，第一个参数都是 `context.Context`
 6. **单向依赖**：下层绝不引用上层类型，公共类型下沉到 `pkg/` 包

 ## 4. 交互层契约（Interaction Layer）
 
 交互层向上暴露给引擎层的接口。
 
 ### 4.1 InteractionService -- 引擎层操作交互层的主入口
 
 ```go
 // InteractionService 交互层服务接口
 // 引擎层通过此接口与用户交互，不直接依赖 Bubble Tea 或终端细节
 type InteractionService interface {
     // StreamResponse 将 LLM 的流式响应推送给用户
     // sessionID: 当前会话 ID，用于多会话场景路由
     // stream: LLM 返回的事件流，交互层负责渲染
     StreamResponse(ctx context.Context, sessionID string, stream <-chan pkg.StreamEvent) error
 
     // RequestConfirmation 请求用户确认（权限弹窗等）
     // 返回用户的选择：true=同意, false=拒绝
     RequestConfirmation(ctx context.Context, sessionID string, prompt ConfirmPrompt) (bool, error)
 
     // ReadInput 阻塞等待用户输入一行文本
     ReadInput(ctx context.Context, sessionID string) (string, error)
 
     // Notify 向用户发送一条通知（非阻塞）
     Notify(ctx context.Context, sessionID string, level NotifyLevel, message string)
 
     // HandleCommand 处理 / 开头的斜杠命令
     // 返回命令是否被处理及处理结果
     HandleCommand(ctx context.Context, sessionID string, input string) (CommandResult, error)
 }
 
 // ConfirmPrompt 确认提示
 type ConfirmPrompt struct {
     Title       string // 简短标题
     Message     string // 详细描述
     ActionType  string // "file_write" / "command" / "tool" / "mcp"
     ActionDetail string // 具体操作描述
 }
 
 // NotifyLevel 通知级别
 type NotifyLevel string
 
 const (
     NotifyInfo  NotifyLevel = "info"
     NotifyWarn  NotifyLevel = "warn"
     NotifyError NotifyLevel = "error"
 )
 
 // CommandResult 命令处理结果
 type CommandResult struct {
     Handled bool   // 是否被识别和处理
     Message string // 反馈消息
     NewSession bool // 是否为创建新会话的命令
 }
 ```
 
 ### 4.2 SessionManager -- 会话生命周期管理
 
 ```go
 // SessionManager 多轮对话会话管理器
 type SessionManager interface {
     // Create 创建新会话
     Create(ctx context.Context, title string) (*pkg.Session, error)
 
     // Get 获取指定会话
     Get(ctx context.Context, sessionID string) (*pkg.Session, error)
 
     // List 列出所有会话
     List(ctx context.Context) ([]*pkg.Session, error)
 
     // AppendMessage 向会话追加一条消息
     AppendMessage(ctx context.Context, sessionID string, msg pkg.Message) error
 
     // SetState 更新会话状态
     SetState(ctx context.Context, sessionID string, state pkg.LoopState) error
 
     // SetPlan 更新会话计划
     SetPlan(ctx context.Context, sessionID string, plan pkg.Plan) error
 
     // Delete 删除会话
     Delete(ctx context.Context, sessionID string) error
 
     // Close 关闭会话（持久化并释放内存缓存）
     Close(ctx context.Context, sessionID string) error
 }
 ```

 ## 5. 引擎层契约（Engine Layer）
 
 引擎层是对外暴露给 cmd 主入口的接口，对内依赖工具层和记忆层。
 
 ### 5.1 Agent -- cmd 入口调用的主接口
 
 ```go
 // Agent 引擎层主入口
 // cmd/supcode/main.go 持有此接口实例，启动交互循环
 type Agent interface {
     // Run 执行一次完整的 Agent Loop
     // input: 用户输入的自然语言指令
     // 返回最终结果和可能的错误
     Run(ctx context.Context, sessionID string, input string) (*AgentResult, error)
 
     // RunPlan 先出计划，等待确认后再执行（Plan Mode）
     RunPlan(ctx context.Context, sessionID string, input string) (*AgentResult, error)
 
     // ApprovePlan 用户确认计划后继续执行
     ApprovePlan(ctx context.Context, sessionID string) (*AgentResult, error)
 
     // GetSession 获取当前会话状态
     GetSession(ctx context.Context, sessionID string) (*pkg.Session, error)
 }
 
 // AgentResult Agent 执行结果
 type AgentResult struct {
     Plan    *pkg.Plan  // 执行的计划
     Summary string     // 结果摘要
     Error   string     // 如有错误
 }
 ```
 
 ### 5.2 LLMClient -- 多供应商 LLM 抽象（核心扩展点）
 
 ```go
 // LLMClient 大语言模型客户端抽象
 // 每个供应商（OpenAI / Claude / DeepSeek / 本地模型）独立实现
 type LLMClient interface {

     // systemPrompt: 系统提示词

     Chat(ctx context.Context, systemPrompt string, messages []pkg.Message, tools []ToolSchema) (<-chan pkg.StreamEvent, error)
 
     // Models 返回可用模型列表
     Models(ctx context.Context) ([]ModelInfo, error)
 
     // ProviderName 返回供应商名称
     ProviderName() string
 }
 
 // ModelInfo 模型信息
 type ModelInfo struct {
     ID          string `json:"id"`
     Name        string `json:"name"`
     MaxTokens   int    `json:"max_tokens"`
     SupportsVision bool `json:"supports_vision"`
 }
 
 // ToolSchema 工具的参数 Schema（JSON Schema 格式，供 LLM function calling）
 type ToolSchema struct {
     Name        string          `json:"name"`
     Description string          `json:"description"`

 ```

 ## 6. 工具层契约（Tool Layer）
 
 工具层对外暴露给引擎层，内部管理内置工具、MCP 外部工具和 Hook 链。
 
 ### 6.1 Tool -- 所有工具的统一接口（核心扩展点）
 
 ```go
 // Tool 工具接口
 // 内置工具、MCP 外部工具、Skill 工具均实现此接口
 type Tool interface {
     // Name 工具名，唯一标识
     Name() string
 
     // Description 工具描述，供 LLM 理解用途
     Description() string
 
     // Schema 工具的参数 JSON Schema，供 LLM function calling
     Schema() ToolSchema
 
     // Execute 执行工具
     Execute(ctx context.Context, params json.RawMessage) (pkg.ToolResult, error)
 }
 ```
 
 ### 6.2 ToolRegistry -- 工具注册中心
 
 ```go
 // ToolRegistry 工具注册中心
 // 引擎层持有此接口，通过它发现和调用工具
 type ToolRegistry interface {
     // Register 注册一个工具
     Register(tool Tool) error
 
     // Unregister 移除一个工具
     Unregister(name string) error
 
     // Get 按名称获取工具
     Get(name string) (Tool, error)
 
     // List 列出所有已注册的工具名
     List() []string
 
     // ListSchemas 列出所有工具的 Schema（供 LLM function calling）
     ListSchemas() []ToolSchema
 
     // Execute 按名称执行工具（会经过 Hook 链和权限检查）
     Execute(ctx context.Context, name string, params json.RawMessage) (pkg.ToolResult, error)
 
     // RegisterHook 注册一个全局钩子
     RegisterHook(hook ToolHook) error
 
     // UnregisterHook 移除一个钩子
     UnregisterHook(hookName string) error
 }
 ```
 
 ### 6.3 ToolHook -- 工具钩子接口
 
 ```go
 // ToolHook 工具钩子
 type ToolHook interface {
     // Name 钩子名
     Name() string
 
     // BeforeTool 工具执行前调用
     // 返回 nil 继续执行；返回 error 中断执行
     BeforeTool(ctx context.Context, toolName string, params json.RawMessage) (json.RawMessage, error)
 
     // AfterTool 工具执行后调用
     AfterTool(ctx context.Context, toolName string, params json.RawMessage, result pkg.ToolResult) error
 }
 ```
 
 ### 6.4 MCPClient -- MCP 外部工具客户端（Phase 3）
 
 ```go
 // MCPClient MCP 协议客户端

     // Connect 连接一个 MCP 服务
 ```

 ## 7. 记忆层契约（Memory Layer）

 ### 7.1 ContextManager -- 上下文管理器

 ```go
 type ContextManager interface {

     TokenCount(ctx context.Context, sessionID string) (int, error)

     ShouldCompress(ctx context.Context, sessionID string) (bool, error)

     // 返回压缩是否成功启动
     Compress(ctx context.Context, sessionID string) error
 
     // GetMemoryCards 获取当前会话的记忆卡片
 ```

 ## 8. 安全层契约（Security Layer）
 
 安全层以横切方式嵌入工具层和引擎层，提供权限决策和工作区隔离。
 
 ### 8.1 PermissionEngine -- 权限引擎
 
 ```go
 // PermissionEngine 权限引擎
 // 嵌入在 ToolRegistry.Execute 的调用链中
 type PermissionEngine interface {
     // Check 检查操作是否允许
     // 返回 allow / ask（需回调交互层确认） / deny
     Check(ctx context.Context, action pkg.Action) (pkg.Decision, error)
 
     // AddRule 动态添加权限规则
     AddRule(rule PermissionRule) error
 
     // RemoveRule 移除权限规则
     RemoveRule(ruleID string) error
 
     // ListRules 列出所有规则
     ListRules() []PermissionRule
 
     // LogAction 记录操作审计日志
     LogAction(ctx context.Context, action pkg.Action, decision pkg.Decision, result string) error
 }
 
 // PermissionRule 权限规则
 type PermissionRule struct {
     ID       string        `json:"id"`
     Scope    string        `json:"scope"`    // "command" / "path" / "tool" / "mcp"
     Pattern  string        `json:"pattern"`  // 匹配模式（正则或前缀）
     Decision pkg.Decision  `json:"decision"` // allow / ask / deny
     Priority int           `json:"priority"` // 优先级，数字越小越优先
 }
 ```
 
 ### 8.2 ConfirmCallback -- 确认回调（引擎层实现，安全层调用）
 
 ```go
 // ConfirmCallback 确认回调函数签名

 // 引擎层在初始化时注入，内部调用 InteractionService.RequestConfirmation
 ```

 ## 9. 配置契约（横切关注点）
 
 ### 9.1 Config -- 配置管理器
 
 ```go
 // Config 配置管理器
 type Config interface {
     // Load 加载配置（合并所有来源）

     // GetString/GetInt/GetBool 类型安全的取值

     AllSettings() map[string]any
 ```

 ### 9.2 配置键常量
 
 ```go
 // 所有配置键的常量定义，避免硬编码字符串
 const (
     ConfigKeyLLMProvider       = "llm.provider"
     ConfigKeyLLMModel          = "llm.model"
     ConfigKeyLLMAPIKey         = "llm.api_key"
     ConfigKeyLLMBaseURL        = "llm.base_url"
 ```

 ## 11. 错误类型契约
 
 ### 11.1 错误分类
 
 ```go
 // 所有跨层传递的错误使用以下类型，方便上层按类型处理
 
 // ErrRetryable 可重试错误
 type ErrRetryable struct {
     Cause error
     RetryAfter time.Duration
 }
 func (e *ErrRetryable) Error() string { return fmt.Sprintf("retryable: %v (after %v)", e.Cause, e.RetryAfter) }
 
 // ErrPermissionDenied 权限拒绝
 type ErrPermissionDenied struct {
     Action pkg.Action
     Rule   string
 }
 func (e *ErrPermissionDenied) Error() string { return fmt.Sprintf("permission denied: %s by rule %s", e.Action.Type, e.Rule) }
 
 // ErrToolNotFound 工具不存在
 type ErrToolNotFound struct {
     ToolName string
 }
 func (e *ErrToolNotFound) Error() string { return fmt.Sprintf("tool not found: %s", e.ToolName) }
 
 // ErrContextExceeded Token 超限
 type ErrContextExceeded struct {
     Current int
     Limit   int
 }
 func (e *ErrContextExceeded) Error() string { return fmt.Sprintf("context exceeded: %d/%d", e.Current, e.Limit) }
 
 // ErrLLMUnavailable LLM 不可用
 type ErrLLMUnavailable struct {
     Provider string
     Cause    error
 }
 func (e *ErrLLMUnavailable) Error() string { return fmt.Sprintf("llm unavailable [%s]: %v", e.Provider, e.Cause) }
 ```
 
 ### 11.2 错误处理约定
 
 | 约定 | 说明 |
 |------|------|
 | 错误包装 | 下层错误向上传递时使用 `fmt.Errorf("...: %w", err)`，保留原始错误链 |
 | 错误日志 | 错误在产生处打 `slog.Error`，上层不再重复打日志 |
 | Sentinel Error | 不可恢复的错误使用 `var ErrXxx = errors.New(...)` 哨兵错误 |
 | 错误判断 | 上层使用 `errors.Is` / `errors.As` 判断，不依赖字符串匹配 |

 ## 13. 子 Agent 协作开发指南
 
 ### 13.1 并行开发策略
 
 由于所有层间交互均通过 interface 定义，子 Agent 可并行开发不同层：
 
 ```
 子Agent A: 交互层 (tui)  -- 依赖 InteractionService + SessionManager interface 定义
 子Agent B: 引擎层 (agent) -- 依赖 LLMClient + ToolRegistry + ContextManager interface
                           -- 提供 Agent interface 给 A
 子Agent C: 工具层 (tools) -- 依赖 Tool + ToolRegistry interface 定义
                           -- 提供 ToolRegistry interface 给 B
 子Agent D: 记忆层 (memory)-- 依赖 MemoryStore + ContextManager interface 定义
                           -- 提供 ContextManager interface 给 B
 子Agent E: 安全层 (perm)  -- 依赖 PermissionEngine interface 定义
                           -- 嵌入在 C 的调用链中
 ```
 
 ### 13.2 集成顺序
 
 1. 先实现所有 mock（每个子 Agent 用 mock 测试自己的模块）
 2. 集成按从下到上：安全层 -> 记忆层 -> 工具层 -> 引擎层 -> 交互层
 3. 每集成一层，运行该层 + 下方所有层的集成测试
 
 ### 13.3 契约版本管理
 
 - `pkg/types.go` 和 `pkg/interfaces.go` 是契约的"单一真相源"
 - 修改 interface 签名的 PR 必须同时更新所有实现者和 mock
 - 向后不兼容的修改需要在 CHANGELOG 中标注 `BREAKING`
